flowchart TD
    START((Start: User Visits System)) --> REGISTER{New User?}
    
    REGISTER -->|Yes| REG[Register/Email Verification]
    REGISTER -->|No| LOGIN[Login]
    REG --> SYNC[Sync User to MongoDB]
    SYNC --> LOGIN
    
    LOGIN --> AUTH{User Type}
    AUTH -->|User/Admin| CLERK[Clerk Authentication]
    AUTH -->|Collector| JWT[JWT Authentication]
    
    CLERK --> ROLE{Check Role}
    ROLE -->|User| USER_DASH[User Dashboard]
    ROLE -->|Admin| ADMIN_DASH[Admin Dashboard]
    JWT --> COL_DASH[Collector Dashboard]
    
    subgraph USER ["User Path"]
        USER_DASH --> USER_CHOICE{Action}
        USER_CHOICE -->|View| SCHEDULE[View Schedule - 33 Wards]
        USER_CHOICE -->|Submit| SUBMIT_REPORT[Submit Waste Report]
        USER_CHOICE -->|Track| MY_REPORTS[View My Reports]
        USER_CHOICE -->|Manage| PROFILE[Update Profile/Ward]
        
        SUBMIT_REPORT --> UPLOAD[Upload Images - Max 5]
        UPLOAD --> CLOUD[Upload to Cloudinary]
        CLOUD --> DESC[Add Description & GPS Location]
        DESC --> ASSIGN[Auto-Assign Collector by Ward]
        ASSIGN --> SAVE_REPORT[Save Report - Status: in-progress]
        SAVE_REPORT --> DB[(MongoDB: Reports)]
        
        PROFILE --> WARD_UPDATE[Update Ward Info]
        WARD_UPDATE --> INSTANT_EMAIL[Send Instant Schedule Reminder]
    end
    
    subgraph COLLECTOR ["Collector Path"]
        COL_DASH --> COL_CHOICE{Action}
        COL_CHOICE -->|Tasks| VIEW_TASKS[View Today Ward Tasks]
        COL_CHOICE -->|Reports| VIEW_REPORTS[View Assigned Reports]
        
        VIEW_TASKS --> AUTO_CREATE[Auto-Create Tasks if Not Exist]
        AUTO_CREATE --> START_TASK[Start Task - Status: in-progress]
        START_TASK --> END_TASK[Complete Task - Status: completed]
        END_TASK --> INCREMENT[Increment Total Collections]
        INCREMENT --> TASK_NOTIFY[Send Email - Ward Users Notified of Pickup Completion]
        
        VIEW_REPORTS --> UPDATE_REP_STATUS[Update Report Status]
        UPDATE_REP_STATUS --> RESOLVE[Mark as Resolved - Status: pending-verification]
        RESOLVE --> WAIT_ADMIN[Awaits Admin Verification]
    end
    
    subgraph ADMIN ["Admin Path"]
        ADMIN_DASH --> ADMIN_CHOICE{Management}
        ADMIN_CHOICE -->|Users| MANAGE_USERS[View All Users & Reports Stats]
        ADMIN_CHOICE -->|Collectors| MANAGE_COL[Manage Collectors - Max 6]
        ADMIN_CHOICE -->|Reports| MANAGE_REPORTS[Review All Reports]
        ADMIN_CHOICE -->|Analytics| VIEW_ANALYTICS[View System Analytics]
        ADMIN_CHOICE -->|Bins| MANAGE_BINS[Manage Smart Bin Status]
        
        MANAGE_COL --> COL_ACTION{Action}
        COL_ACTION -->|Create| NEW_COL[Create Collector - Min 5 Wards]
        COL_ACTION -->|Update| EDIT_COL[Edit Collector Details]
        COL_ACTION -->|Delete| DEL_COL[Delete Collector]
        
        MANAGE_REPORTS --> ALL_REPORTS[View All Reports]
        ALL_REPORTS --> VERIFY{Assess Report}
        VERIFY -->|Approve| APPROVE_REP[Report Status: resolved]
        VERIFY -->|Reject| REJECT_REP[Report Sent Back to in-progress]
        APPROVE_REP --> SEND_RES_EMAIL[Send Report Resolution Email to User]
        REJECT_REP --> BACK_COLLECTOR[Back to Collector for Re-work]
    end
    
    subgraph AUTOMATED ["Automated Services"]
        CRON[Cron Job - Daily 6 PM & 9 PM NPT] --> CHECK_SCHEDULE[Check Tomorrow's Pickup Schedule]
        CHECK_SCHEDULE --> FIND_USERS[Find All Users in Scheduled Wards]
        FIND_USERS --> DUP_CHECK{Already Reminded Today?}
        DUP_CHECK -->|No| REMIND[Send Schedule Reminder Email]
        DUP_CHECK -->|Yes| SKIP[Skip - Prevent Duplicate]
        REMIND --> DUP_CHECK
        
        BIN_ALERT[Smart Bin Status Change] --> CHECK_BIN{Status Changed?}
        CHECK_BIN -->|Full| FULL_ALERT[Send Bin Full Alert to Users/Collectors/Admins]
        CHECK_BIN -->|Emptied| EMPTY_ALERT[Send Bin Emptied Alert to Users/Collectors/Admins]
        CHECK_BIN -->|No Change| SKIP_ALERT[Skip - No Status Change]
    end
    
    subgraph EMAIL ["Email Notification System"]
        EMAIL1[Email Verification - Clerk]
        EMAIL2[Schedule Reminder Email]
        EMAIL3[Pickup Completion Email]
        EMAIL4[Report Resolution Email]
        EMAIL5[Bin Status Alerts]
        EMAIL_CONFIG[Provider: Resend / SMTP / Console]
    end
    
    subgraph DATABASE_GROUP ["MongoDB Database & Storage"]
        USERS[(Users Collection)]
        COLLECTORS[(Collectors Collection)]
        REPORTS[(Reports Collection)]
        TASKS[(Ward Tasks)]
        BINS[(Bin Alerts)]
        CLOUDINARY[Cloudinary - Image Storage]
    end
    
    INSTANT_EMAIL --> EMAIL2
    TASK_NOTIFY --> EMAIL3
    SEND_RES_EMAIL --> EMAIL4
    REMIND --> EMAIL2
    FULL_ALERT --> EMAIL5
    EMPTY_ALERT --> EMAIL5
    
    SAVE_REPORT --> REPORTS
    DB --> REPORTS
    ASSIGN --> REPORTS
    NEW_COL --> COLLECTORS
    EDIT_COL --> COLLECTORS
    AUTO_CREATE --> TASKS
    START_TASK --> TASKS
    END_TASK --> TASKS
    WAIT_ADMIN --> REPORTS
    VIEW_TASKS --> TASKS
    CHECK_SCHEDULE --> TASKS
    MANAGE_BINS --> BINS
    BIN_ALERT --> BINS
    CHECK_BIN --> BINS
    UPLOAD --> CLOUDINARY
    CLOUD --> CLOUDINARY
    SYNC --> USERS
    MANAGE_USERS --> USERS
    PROFILE --> USERS
    
    EMAIL3 --> EMAIL_CONFIG
    EMAIL2 --> EMAIL_CONFIG
    EMAIL4 --> EMAIL_CONFIG
    EMAIL5 --> EMAIL_CONFIG
    
    EMAIL_CONFIG --> SUCCESS{All Operations Complete?}
    SUCCESS -->|Yes| END((End: System Operations Complete))
    SUCCESS -->|No| START
